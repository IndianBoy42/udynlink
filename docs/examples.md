## Minimal Host: Loading and Calling a C Module

This example shows the smallest possible host firmware that can load a C module, call an exported function, and unload it.

### Module source (`mod_hello.c`)

```c
#include <stdio.h>

int hello(int arg) {
    printf("Hello World! arg=%d\n", arg);
    return -arg;
}

int test(void) {
    return (hello(0) == 0 && hello(10) == -10);
}
```

Compile the module with the toolchain:

```bash
cd scripts
python3 mkmodule --gen-c-header --header-path ../host_firmware \
    ../modules/mod_hello.c
```

This produces `mod_hello_module_data.h`, a C header containing the module as a byte array.

### Host firmware (`host_minimal.c`)

```c
#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>
#include <string.h>
#include <stdarg.h>

#define UDYNLINK_MAX_HANDLES 4
#include "udynlink.h"
#include "mod_hello_module_data.h"   /* generated by mkmodule --gen-c-header */

/* -------------------------------------------------------------------------- */
/*  External callbacks required by udynlink                                   */
/* -------------------------------------------------------------------------- */

void *udynlink_external_malloc(size_t size) {
    return malloc(size);
}

void udynlink_external_free(void *p) {
    free(p);
}

void udynlink_external_vprintf(const char *s, va_list va) {
    vprintf(s, va);
}

int udynlink_external_is_pointer_in_ram(const void *p) {
    /* SRAM on this device is 0x20000000 -- 0x20010000 */
    uintptr_t addr = (uintptr_t)p;
    return (addr >= 0x20000000 && addr < 0x20010000);
}

uint32_t udynlink_external_resolve_symbol(const char *name) {
    if (!strcmp(name, "printf"))
        return (uint32_t)(uintptr_t)&printf;
    return 0;
}

uint32_t udynlink_external_resolve_critical_symbol(const char *name) {
    (void)name;
    return 0;
}

struct _udynlink_module_t *udynlink_external_get_module_handle(const char *module_name) {
    (void)module_name;
    return NULL;   /* no dependency tracking in this minimal host */
}

/* -------------------------------------------------------------------------- */
/*  Main application                                                          */
/* -------------------------------------------------------------------------- */

int main(void) {
    udynlink_module_t mod;
    udynlink_error_t err;

    /* Optional: enable debug output during development */
    udynlink_set_debug_level(UDYNLINK_DEBUG_INFO);

    /* Load the module from the embedded byte array */
    err = udynlink_load_module(&mod, mod_hello_module_data, NULL, 0,
                               UDYNLINK_LOAD_MODE_COPY_ALL);
    if (err != UDYNLINK_OK) {
        printf("Load failed: %s\n", udynlink_error_msg(&err));
        return 1;
    }

    /* Before calling any module function, set the LOT base register */
    uint32_t *lot_base = (uint32_t *)UDYNLINK_LOT_BASE_ADDR;
    *lot_base = mod.ram_base;

    /* Look up and call the "test" function */
    udynlink_sym_t sym;
    if (udynlink_lookup_symbol(&mod, "test", &sym) == NULL) {
        printf("Symbol 'test' not found\n");
        udynlink_unload_module(&mod);
        return 1;
    }

    int (*test_fn)(void) = (int (*)(void))(uintptr_t)sym.val;
    int ok = test_fn();
    printf("Module test returned: %d\n", ok);

    /* Unload when done */
    udynlink_unload_module(&mod);
    return 0;
}
```

### Key points

- `UDYNLINK_MAX_HANDLES` **must** be defined before including `udynlink.h`.
- `*(uint32_t *)UDYNLINK_LOT_BASE_ADDR = mod.ram_base` is **mandatory** before every cross-module call. The module's assembly prologue reads this fixed address to set `r9`.
- Every host must implement all seven functions in `udynlink_externals.h`, even if some are stubs.

See also [How It Works](how-it-works.md) and [Host Guide](integrating-as-host.md).

## Host with Symbol Table

For small firmware images, a linear search over a static array of `{name, address}` pairs is sufficient.

### Building the table

`udynlink_externals.h` provides the `UDYNLINK_SYMBOL` macro:

```c
#define UDYNLINK_SYMBOL(sym) { #sym, (void *)(uintptr_t)(sym) }
```

### Host implementation

```c
#include <string.h>
#include <stdint.h>
#include <stdarg.h>
#include <stdio.h>

#define UDYNLINK_MAX_HANDLES 4
#include "udynlink.h"

/* -------------------------------------------------------------------------- */
/*  Symbol table                                                              */
/* -------------------------------------------------------------------------- */

typedef struct {
    const char *name;
    void       *addr;
} host_symbol_t;

static const host_symbol_t host_symbols[] = {
    UDYNLINK_SYMBOL(printf),
    UDYNLINK_SYMBOL(puts),
    UDYNLINK_SYMBOL(memset),
    UDYNLINK_SYMBOL(memcpy),
    UDYNLINK_SYMBOL(malloc),
    UDYNLINK_SYMBOL(free),
    { NULL, NULL }
};

uint32_t udynlink_external_resolve_symbol(const char *name) {
    for (const host_symbol_t *s = host_symbols; s->name != NULL; ++s) {
        if (!strcmp(s->name, name)) {
            return (uint32_t)(uintptr_t)s->addr;
        }
    }
    return 0;   /* not found */
}

uint32_t udynlink_external_resolve_critical_symbol(const char *name) {
    (void)name;
    return 0;   /* defer to linear table */
}

/* -------------------------------------------------------------------------- */
/*  Other required externals (malloc, free, vprintf, is_pointer_in_ram)     */
/* -------------------------------------------------------------------------- */

void *udynlink_external_malloc(size_t size) { return malloc(size); }
void udynlink_external_free(void *p) { free(p); }
void udynlink_external_vprintf(const char *s, va_list va) { vprintf(s, va); }
int udynlink_external_is_pointer_in_ram(const void *p) {
    return ((uintptr_t)p >= 0x20000000 && (uintptr_t)p < 0x20010000);
}
struct _udynlink_module_t *udynlink_external_get_module_handle(const char *name) {
    (void)name; return NULL;
}
```

### Advantages and trade-offs

- **Simple:** No build-time tooling beyond the module compiler.
- **Predictable:** Lookup cost is `O(N)` over the number of exported host symbols.
- **Best for:** Firmware images with fewer than ~100 exported symbols.

For larger symbol sets, switch to the [hash-based approach](#host-with-hash-based-resolution).

## Host with Hash-Based Resolution

For firmware with hundreds of exported symbols, use the `mkhostsyms` script to generate a GNU-hash table for O(1) lookups.

### Step 1: Generate the hash table from the host ELF

After building your host firmware ELF (e.g., `host.elf`), run:

```bash
cd scripts
python3 mkhostsyms \
    --elf ../build/host.elf \
    --output ../host_firmware/host_syms.h
```

The generated `host_syms.h` contains a fully initialized `udynlink_hash_table_t` named `g_host_sym_table`, along with all the backing arrays (Bloom filter, buckets, hash values, addresses, string table).

### Step 2: Use the hash table in the host

```c
#include <stdarg.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define UDYNLINK_MAX_HANDLES 8
#include "udynlink.h"
#include "udynlink_hash.h"
#include "host_syms.h"          /* generated by mkhostsyms */

/* -------------------------------------------------------------------------- */
/*  Hash-based symbol resolution                                              */
/* -------------------------------------------------------------------------- */

uint32_t udynlink_external_resolve_symbol(const char *name) {
    void *addr = udynlink_resolve_hashed_symbol(&g_host_sym_table, name);
    return (addr != NULL) ? (uint32_t)(uintptr_t)addr : 0;
}

uint32_t udynlink_external_resolve_critical_symbol(const char *name) {
    (void)name;
    return 0;   /* all symbols go through the hash table */
}

/* -------------------------------------------------------------------------- */
/*  Other required externals                                                  */
/* -------------------------------------------------------------------------- */

void *udynlink_external_malloc(size_t size) { return malloc(size); }
void udynlink_external_free(void *p) { free(p); }
void udynlink_external_vprintf(const char *s, va_list va) { vprintf(s, va); }
int udynlink_external_is_pointer_in_ram(const void *p) {
    return ((uintptr_t)p >= 0x20000000 && (uintptr_t)p < 0x20010000);
}
struct _udynlink_module_t *udynlink_external_get_module_handle(const char *name) {
    (void)name; return NULL;
}
```

### Filtering symbols

If the host ELF contains many symbols you do not want to expose to modules, use the `--filter` option:

```bash
python3 mkhostsyms \
    --elf ../build/host.elf \
    --output ../host_firmware/host_syms.h \
    --filter "^hal_|^os_"
```

Only symbols matching the regular expression are included in the hash table.

### Advantages

- **O(1) average lookup:** Bloom filter eliminates most negative lookups instantly.
- **Build-time generation:** The hash table is a compile-time constant; no runtime initialization code is needed.
- **Scales well:** Suitable for large firmware images with many APIs.

## XIP (Execute In Place) from Flash

XIP mode copies **only** the `.data` section to RAM; code executes directly from flash. This saves RAM at the cost of slightly slower execution (flash wait states) and the requirement that the code remain valid while the module is loaded.

### When to use XIP

- Flash is large and RAM is scarce.
- The module binary is stored in read-only flash that will not be erased or updated while the module is loaded.
- Execution speed is acceptable with flash wait states.

### Module compilation

Compile normally; the load mode is chosen at runtime by the host:

```bash
cd scripts
python3 mkmodule --gen-c-header --header-path ../host_firmware \
    ../modules/mod_sensor.c
```

### Host setup for XIP

```c
#include <stdint.h>
#include <stdarg.h>
#include <stdio.h>
#include <string.h>

#define UDYNLINK_MAX_HANDLES 4
#include "udynlink.h"
#include "mod_sensor_module_data.h"

void *udynlink_external_malloc(size_t size) { return malloc(size); }
void udynlink_external_free(void *p) { free(p); }
void udynlink_external_vprintf(const char *s, va_list va) { vprintf(s, va); }

/* For XIP, the loader needs to know which addresses are RAM */
int udynlink_external_is_pointer_in_ram(const void *p) {
    uintptr_t addr = (uintptr_t)p;
    /* Example: 128 KiB SRAM at 0x20000000 */
    return (addr >= 0x20000000 && addr < 0x20020000);
}

uint32_t udynlink_external_resolve_symbol(const char *name) {
    if (!strcmp(name, "printf"))  return (uint32_t)(uintptr_t)&printf;
    if (!strcmp(name, "memcpy"))  return (uint32_t)(uintptr_t)&memcpy;
    return 0;
}

uint32_t udynlink_external_resolve_critical_symbol(const char *name) {
    (void)name; return 0;
}

struct _udynlink_module_t *udynlink_external_get_module_handle(const char *name) {
    (void)name; return NULL;
}

int main(void) {
    udynlink_module_t mod;

    /* Assume mod_sensor_module_data is in flash at 0x08010000 */
    const void *flash_addr = (const void *)0x08010000;

    udynlink_error_t err = udynlink_load_module(&mod, flash_addr, NULL, 0,
                                                UDYNLINK_LOAD_MODE_XIP);
    if (err != UDYNLINK_OK) {
        printf("XIP load failed: %s\n", udynlink_error_msg(&err));
        return 1;
    }

    printf("Module loaded. RAM used: %u bytes\n", udynlink_get_ram_size(&mod));

    /* Call an exported function */
    uint32_t *lot_base = (uint32_t *)UDYNLINK_LOT_BASE_ADDR;
    *lot_base = mod.ram_base;

    udynlink_sym_t sym;
    if (udynlink_lookup_symbol(&mod, "read_sensor", &sym)) {
        int (*read_fn)(void) = (int (*)(void))(uintptr_t)sym.val;
        int value = read_fn();
        printf("Sensor value: %d\n", value);
    }

    udynlink_unload_module(&mod);
    return 0;
}
```

### RAM savings comparison

For a module with 1 KiB code, 256 bytes `.data`, 64 bytes `.bss`, and 16 LOT entries:

| Load mode | RAM usage |
|-----------|-----------|
| `COPY_ALL` | 16*4 + 1024 + 256 + 64 = **1392 bytes** |
| `COPY_CODE` | 16*4 + 1024 + 256 + 64 = **1392 bytes** |
| `XIP` | 16*4 + 256 + 64 = **384 bytes** |

XIP saves **1008 bytes** in this example because the 1 KiB code section stays in flash.

## Inter-Module Dependencies

A module can declare dependencies on other modules. The loader enforces that all dependencies are loaded first, and it resolves `extern` symbols by searching dependency modules before falling back to the host.

### Provider module (`mod_provider.c`)

```c
/* Exports utility functions for other modules */
int provider_add(int a, int b) {
    return a + b;
}

int provider_mul(int a, int b) {
    return a * b;
}
```

Compile **without** `--depends` because this is the root provider:

```bash
python3 mkmodule --gen-c-header --header-path ../host_firmware \
    ../modules/mod_provider.c
```

### Consumer module (`mod_consumer.c`)

```c
/* Calls functions from mod_provider */
extern int provider_add(int a, int b);
extern int provider_mul(int a, int b);

int test(void) {
    return (provider_add(2, 3) == 5 && provider_mul(4, 5) == 20);
}
```

Compile **with** `--depends provider`:

```bash
python3 mkmodule --gen-c-header --header-path ../host_firmware \
    --depends provider \
    ../modules/mod_consumer.c
```

The module name used in `--depends` must match the module's own name symbol (derived from the first source file name or `--name`).

### Host code loading both modules

```c
#include <stdint.h>
#include <string.h>
#include <stdio.h>
#include <stdarg.h>

#define UDYNLINK_MAX_HANDLES 8
#include "udynlink.h"
#include "mod_provider_module_data.h"
#include "mod_consumer_module_data.h"

/* -------------------------------------------------------------------------- */
/*  Minimal externals                                                         */
/* -------------------------------------------------------------------------- */

static udynlink_module_t *g_modules[UDYNLINK_MAX_HANDLES];
static int g_module_count = 0;

void *udynlink_external_malloc(size_t size) { return malloc(size); }
void udynlink_external_free(void *p) { free(p); }
void udynlink_external_vprintf(const char *s, va_list va) { vprintf(s, va); }
int udynlink_external_is_pointer_in_ram(const void *p) {
    return ((uintptr_t)p >= 0x20000000 && (uintptr_t)p < 0x20010000);
}

uint32_t udynlink_external_resolve_symbol(const char *name) {
    if (!strcmp(name, "printf")) return (uint32_t)(uintptr_t)&printf;
    return 0;
}

uint32_t udynlink_external_resolve_critical_symbol(const char *name) {
    (void)name; return 0;
}

/* The loader uses this to verify that declared dependencies are loaded */
struct _udynlink_module_t *udynlink_external_get_module_handle(const char *module_name) {
    for (int i = 0; i < g_module_count; i++) {
        if (g_modules[i] == NULL) continue;
        const char *name = udynlink_get_module_name(g_modules[i]);
        if (name && !strcmp(name, module_name))
            return g_modules[i];
    }
    return NULL;
}

static void register_module(udynlink_module_t *p_mod) {
    if (g_module_count < UDYNLINK_MAX_HANDLES)
        g_modules[g_module_count++] = p_mod;
}

static void unregister_module(udynlink_module_t *p_mod) {
    for (int i = 0; i < g_module_count; i++) {
        if (g_modules[i] == p_mod) {
            g_modules[i] = g_modules[--g_module_count];
            return;
        }
    }
}

/* -------------------------------------------------------------------------- */
/*  Main application                                                          */
/* -------------------------------------------------------------------------- */

int main(void) {
    udynlink_module_t mod_provider;
    udynlink_module_t mod_consumer;
    udynlink_error_t err;

    /* 1. Load the provider FIRST */
    memset(&mod_provider, 0, sizeof(mod_provider));
    err = udynlink_load_module(&mod_provider, mod_provider_module_data,
                               NULL, 0, UDYNLINK_LOAD_MODE_COPY_ALL);
    if (err != UDYNLINK_OK) {
        printf("Provider load failed: %s\n", udynlink_error_msg(&err));
        return 1;
    }
    register_module(&mod_provider);

    /* 2. Load the consumer (dependencies are validated automatically) */
    memset(&mod_consumer, 0, sizeof(mod_consumer));
    err = udynlink_load_module(&mod_consumer, mod_consumer_module_data,
                               NULL, 0, UDYNLINK_LOAD_MODE_COPY_ALL);
    if (err != UDYNLINK_OK) {
        printf("Consumer load failed: %s\n", udynlink_error_msg(&err));
        udynlink_unload_module(&mod_provider);
        return 1;
    }
    register_module(&mod_consumer);

    /* 3. Call the consumer */
    uint32_t *lot_base = (uint32_t *)UDYNLINK_LOT_BASE_ADDR;
    *lot_base = mod_consumer.ram_base;

    udynlink_sym_t sym;
    if (udynlink_lookup_symbol(&mod_consumer, "test", &sym)) {
        int (*test_fn)(void) = (int (*)(void))(uintptr_t)sym.val;
        printf("Consumer test: %d\n", test_fn());
    }

    /* 4. Unload in REVERSE order: consumer first, then provider */
    unregister_module(&mod_consumer);
    udynlink_unload_module(&mod_consumer);

    unregister_module(&mod_provider);
    udynlink_unload_module(&mod_provider);

    return 0;
}
```

### Unload order rules

- A module with `dep_refcount > 0` cannot be unloaded (`UDYNLINK_ERR_MODULE_IN_USE`).
- Always unload **consumers** before **providers**.
- If you try to unload `mod_provider` while `mod_consumer` is still loaded, the loader will reject it.

## C++ Module with Constructors

udynlink supports C++ modules compiled with `-fno-exceptions -fno-rtti -fno-use-cxa-atexit`. Global constructors are collected in `__init_array` and must be executed by the host after loading.

### C++ module (`mod_hello_cpp.cpp`)

```cpp
#include <stdio.h>

class Test {
public:
    Test(const char* arg) {
        printf("%s_constructor\n", arg);
    }
};

/* Global instance -- constructor runs when __init_array is invoked */
static Test constructor_test("static");

extern "C" int hello(int arg) {
    printf("Hello World! arg=%d\n", arg);
    return -arg;
}

extern "C" int test(void) {
    return (hello(0) == 0 && hello(10) == -10);
}
```

Compile with the toolchain (automatically adds C++ flags and `cpp_init_fini.c`):

```bash
python3 mkmodule --gen-c-header --header-path ../host_firmware \
    ../modules/mod_hello_cpp.cpp
```

### Host code

```c
#include <stdint.h>
#include <stdio.h>
#include <stdarg.h>

#define UDYNLINK_MAX_HANDLES 4
#include "udynlink.h"
#include "mod_hello_cpp_module_data.h"

void *udynlink_external_malloc(size_t size) { return malloc(size); }
void udynlink_external_free(void *p) { free(p); }
void udynlink_external_vprintf(const char *s, va_list va) { vprintf(s, va); }
int udynlink_external_is_pointer_in_ram(const void *p) {
    return ((uintptr_t)p >= 0x20000000 && (uintptr_t)p < 0x20010000);
}

uint32_t udynlink_external_resolve_symbol(const char *name) {
    if (!strcmp(name, "printf")) return (uint32_t)(uintptr_t)&printf;
    return 0;
}

uint32_t udynlink_external_resolve_critical_symbol(const char *name) {
    (void)name; return 0;
}

struct _udynlink_module_t *udynlink_external_get_module_handle(const char *name) {
    (void)name; return NULL;
}

int main(void) {
    udynlink_module_t mod;

    udynlink_error_t err = udynlink_load_module(&mod, mod_hello_cpp_module_data,
                               NULL, 0, UDYNLINK_LOAD_MODE_COPY_ALL);
    if (err != UDYNLINK_OK) {
        printf("Load failed: %s\n", udynlink_error_msg(&err));
        return 1;
    }

    /* CRITICAL: Set LOT base BEFORE calling udynlink_cpp_init, because
       constructors may touch module data through the LOT. */
    uint32_t *lot_base = (uint32_t *)UDYNLINK_LOT_BASE_ADDR;
    *lot_base = mod.ram_base;

    /* Run global constructors */
    udynlink_cpp_init(&mod);

    /* Now safe to call regular module functions */
    *lot_base = mod.ram_base;   /* re-set in case another module ran */

    udynlink_sym_t sym;
    if (udynlink_lookup_symbol(&mod, "test", &sym)) {
        int (*test_fn)(void) = (int (*)(void))(uintptr_t)sym.val;
        printf("C++ test result: %d\n", test_fn());
    }

    udynlink_unload_module(&mod);
    return 0;
}
```

### Constructor execution sequence

1. Host calls `udynlink_load_module`.
2. Loader copies sections, applies relocations, resolves externals.
3. Host writes `ram_base` to `UDYNLINK_LOT_BASE_ADDR`.
4. Host calls `udynlink_cpp_init(&mod)`.
5. `udynlink_cpp_init` looks up `__init_array`, sets LOT base again internally, and calls the constructor table function.
6. All global `static` C++ objects are now initialized.
7. Host may call regular exported functions.

## Streaming Load from SD Card

Use `udynlink_load_module_stream` when the module image is too large to buffer entirely in RAM (e.g., stored on an SD card as a file).

### FatFS I/O adapter

```c
#include "ff.h"                 /* FatFS header */
#include "udynlink.h"

/* -------------------------------------------------------------------------- */
/*  FatFS streaming I/O callbacks                                             */
/* -------------------------------------------------------------------------- */

typedef struct {
    FIL fil;
} fatfs_ctx_t;

static int32_t fatfs_read(void *pv_ctx, void *buf, uint32_t num_bytes, uint32_t offset) {
    fatfs_ctx_t *ctx = (fatfs_ctx_t *)pv_ctx;
    FRESULT res;
    UINT br;

    /* Seek to the requested offset */
    res = f_lseek(&ctx->fil, offset);
    if (res != FR_OK) return -1;

    res = f_read(&ctx->fil, buf, num_bytes, &br);
    if (res != FR_OK) return -1;

    return (int32_t)br;
}

static int32_t fatfs_get_size(void *pv_ctx) {
    fatfs_ctx_t *ctx = (fatfs_ctx_t *)pv_ctx;
    return (int32_t)f_size(&ctx->fil);
}

/* -------------------------------------------------------------------------- */
/*  Load module from "module.bin" on SD card                                  */
/* -------------------------------------------------------------------------- */

int load_module_from_sd(const char *path, udynlink_module_t *p_mod) {
    fatfs_ctx_t ctx;
    FRESULT res;

    res = f_open(&ctx.fil, path, FA_READ);
    if (res != FR_OK) {
        printf("Failed to open %s\n", path);
        return -1;
    }

    udynlink_io_t io = {
        .read = fatfs_read,
        .get_size = fatfs_get_size,
        .pv_ctx = &ctx
    };

    /* Query metadata size for work buffer */
    uint32_t work_size = udynlink_get_stream_metadata_size(&io);
    if (work_size == 0) {
        printf("Invalid module or I/O error\n");
        f_close(&ctx.fil);
        return -1;
    }

    /* Allocate work buffer (can also use a static pool) */
    void *work_buf = udynlink_external_malloc(work_size);
    if (!work_buf) {
        printf("Out of memory for work buffer\n");
        f_close(&ctx.fil);
        return -1;
    }

    /* Optional: query RAM requirements before allocating module RAM */
    uint32_t ram_needed = udynlink_get_ram_requirements_stream(&io, UDYNLINK_LOAD_MODE_COPY_ALL);
    printf("Module needs %u bytes of RAM\n", ram_needed);

    /* Load the module */
    udynlink_error_t err = udynlink_load_module_stream(
        p_mod, &io, NULL, 0,          /* auto-allocate RAM */
        UDYNLINK_LOAD_MODE_COPY_ALL,
        work_buf, work_size);

    f_close(&ctx.fil);
    udynlink_external_free(work_buf);

    if (err != UDYNLINK_OK) {
        printf("Stream load failed: %s\n", udynlink_error_msg(&err));
        return -1;
    }

    return 0;
}
```

### Work buffer sizing

- **Minimum:** `64` bytes (`UDYNLINK_STREAM_MIN_WORK_BUF_SIZE`).
- **Optimal:** Use `udynlink_get_stream_metadata_size()` to get a size that allows the loader to read the entire header + relocation table + symbol table in a single `read()` call. This reduces SD-card access overhead.
- The work buffer is only a scratch area for partial reads; it does **not** need to persist after loading.

### Streaming limitations

- **XIP is not supported** (`UDYNLINK_LOAD_MODE_XIP` returns `UDYNLINK_ERR_LOAD_UNABLE_TO_XIP`).
- `COPY_CODE` mode is internally converted to `COPY_ALL` because the header must be present in RAM for relocation processing.

## Multiple Module Instances

The eh2k fork allows loading **multiple instances of the same module**. Each instance gets its own RAM, LOT, and data sections. This is useful when the same algorithmic code needs to operate on different state.

### Example: two independent counters

Module source (`mod_counter.c`):

```c
static int counter = 0;

int increment(void) {
    return ++counter;
}

int get_count(void) {
    return counter;
}
```

### Host code loading two instances

```c
#include <stdint.h>
#include <stdio.h>
#include <string.h>

#define UDYNLINK_MAX_HANDLES 8
#include "udynlink.h"
#include "mod_counter_module_data.h"

void *udynlink_external_malloc(size_t size) { return malloc(size); }
void udynlink_external_free(void *p) { free(p); }
void udynlink_external_vprintf(const char *s, va_list va) { vprintf(s, va); }
int udynlink_external_is_pointer_in_ram(const void *p) {
    return ((uintptr_t)p >= 0x20000000 && (uintptr_t)p < 0x20010000);
}
uint32_t udynlink_external_resolve_symbol(const char *name) {
    (void)name; return 0;
}
uint32_t udynlink_external_resolve_critical_symbol(const char *name) {
    (void)name; return 0;
}
struct _udynlink_module_t *udynlink_external_get_module_handle(const char *name) {
    (void)name; return NULL;
}

int main(void) {
    udynlink_module_t mod_a;
    udynlink_module_t mod_b;

    /* Load instance A */
    memset(&mod_a, 0, sizeof(mod_a));
    udynlink_error_t err = udynlink_load_module(&mod_a, mod_counter_module_data,
                               NULL, 0, UDYNLINK_LOAD_MODE_COPY_ALL);
    if (err != UDYNLINK_OK) {
        printf("Load A failed: %s\n", udynlink_error_msg(&err));
        return 1;
    }

    /* Load instance B */
    memset(&mod_b, 0, sizeof(mod_b));
    err = udynlink_load_module(&mod_b, mod_counter_module_data,
                               NULL, 0, UDYNLINK_LOAD_MODE_COPY_ALL);
    if (err != UDYNLINK_OK) {
        printf("Load B failed: %s\n", udynlink_error_msg(&err));
        udynlink_unload_module(&mod_a);
        return 1;
    }

    uint32_t *lot_base = (uint32_t *)UDYNLINK_LOT_BASE_ADDR;

    /* Call increment() on instance A */
    *lot_base = mod_a.ram_base;
    udynlink_sym_t sym;
    udynlink_lookup_symbol(&mod_a, "increment", &sym);
    int (*inc)(void) = (int (*)(void))(uintptr_t)sym.val;
    printf("A increment: %d\n", inc());   /* 1 */
    printf("A increment: %d\n", inc());   /* 2 */

    /* Call increment() on instance B */
    *lot_base = mod_b.ram_base;
    udynlink_lookup_symbol(&mod_b, "increment", &sym);
    inc = (int (*)(void))(uintptr_t)sym.val;
    printf("B increment: %d\n", inc());   /* 1 (independent state) */
    printf("B increment: %d\n", inc());   /* 2 */

    /* Verify A's state is untouched */
    *lot_base = mod_a.ram_base;
    udynlink_lookup_symbol(&mod_a, "get_count", &sym);
    int (*get)(void) = (int (*)(void))(uintptr_t)sym.val;
    printf("A count: %d\n", get());       /* 2 */

    udynlink_unload_module(&mod_b);
    udynlink_unload_module(&mod_a);
    return 0;
}
```

### Critical rule

**Always re-write `UDYNLINK_LOT_BASE_ADDR` before switching to another module instance.** The LOT base is global state; if you forget to update it, the new instance will use the old instance's data pointers.

## Embedding Modules in Firmware

Modules can be embedded directly into the host firmware as constant C byte arrays, eliminating the need for a filesystem or external storage.

### Generate the C header

```bash
cd scripts
python3 mkmodule --gen-c-header --header-path ../host_firmware \
    --target cortex-m4 \
    ../modules/mod_hello.c
```

This creates `mod_hello_module_data.h`:

```c
#ifndef MOD_HELLO_MODULE_DATA_H
#define MOD_HELLO_MODULE_DATA_H

static const unsigned char mod_hello_module_data[] = {
    0x55, 0x44, 0x4C, 0x4D,   /* "UDLM" signature */
    /* ... rest of module binary ... */
};

#endif
```

### Include and load

```c
#include "udynlink.h"
#include "mod_hello_module_data.h"

int main(void) {
    udynlink_module_t mod;

    /* The array is in flash; load into RAM */
    udynlink_error_t err = udynlink_load_module(
        &mod,
        mod_hello_module_data,   /* base_addr in flash */
        NULL, 0,                /* auto-allocate RAM */
        UDYNLINK_LOAD_MODE_COPY_ALL);

    /* ... use module ... */
}
```

### Multiple modules

Repeat `--gen-c-header` for each module:

```bash
python3 mkmodule --gen-c-header --header-path ../host_firmware \
    ../modules/mod_hello.c
python3 mkmodule --gen-c-header --header-path ../host_firmware \
    ../modules/mod_sensor.c
```

Then include both headers and load whichever module you need at runtime.

### When to embed

- Firmware has dedicated flash space for modules.
- No external storage (SD card, SPI flash) is available.
- Modules are small enough to fit in the internal flash budget.
- Over-the-air updates are performed on the whole firmware image, not individual modules.

## Pre-Computing RAM Requirements

Instead of auto-allocating with `udynlink_external_malloc`, the host can pre-compute how much RAM a module needs and provide a fixed pool or slab. This avoids fragmentation and gives deterministic memory usage.

### Using `udynlink_get_ram_requirements`

```c
#include <stdint.h>
#include <stdio.h>
#include <string.h>

#define UDYNLINK_MAX_HANDLES 4
#include "udynlink.h"
#include "mod_hello_module_data.h"

/* Simple memory pool */
static uint8_t module_pool[8192];
static uint32_t pool_offset = 0;

void *pool_malloc(size_t size) {
    size = (size + 3) & ~3U;   /* align to 4 bytes */
    if (pool_offset + size > sizeof(module_pool))
        return NULL;
    void *p = &module_pool[pool_offset];
    pool_offset += size;
    return p;
}

void pool_free(void *p) {
    (void)p;
    /* Simple pool: no individual frees; reset pool_offset to free all */
}

void *udynlink_external_malloc(size_t size) { return pool_malloc(size); }
void udynlink_external_free(void *p) { pool_free(p); }
void udynlink_external_vprintf(const char *s, va_list va) { vprintf(s, va); }
int udynlink_external_is_pointer_in_ram(const void *p) {
    return ((uintptr_t)p >= 0x20000000 && (uintptr_t)p < 0x20010000);
}
uint32_t udynlink_external_resolve_symbol(const char *name) {
    if (!strcmp(name, "printf")) return (uint32_t)(uintptr_t)&printf;
    return 0;
}
uint32_t udynlink_external_resolve_critical_symbol(const char *name) {
    (void)name; return 0;
}
struct _udynlink_module_t *udynlink_external_get_module_handle(const char *name) {
    (void)name; return NULL;
}

int main(void) {
    /* Query RAM needed before loading */
    uint32_t ram_needed = udynlink_get_ram_requirements(
        mod_hello_module_data, UDYNLINK_LOAD_MODE_COPY_ALL);

    printf("Module needs %u bytes of RAM\n", ram_needed);

    if (ram_needed > sizeof(module_pool) - pool_offset) {
        printf("Not enough pool space!\n");
        return 1;
    }

    /* Provide the pool as the load address */
    udynlink_module_t mod;
    void *load_addr = &module_pool[pool_offset];

    udynlink_error_t err = udynlink_load_module(
        &mod, mod_hello_module_data,
        load_addr, sizeof(module_pool) - pool_offset,
        UDYNLINK_LOAD_MODE_COPY_ALL);

    if (err != UDYNLINK_OK) {
        printf("Load failed: %s\n", udynlink_error_msg(&err));
        return 1;
    }

    printf("Loaded at pool offset %u, RAM size %u\n",
           pool_offset, udynlink_get_ram_size(&mod));

    /* Use module ... */
    uint32_t *lot_base = (uint32_t *)UDYNLINK_LOT_BASE_ADDR;
    *lot_base = mod.ram_base;

    udynlink_sym_t sym;
    if (udynlink_lookup_symbol(&mod, "test", &sym)) {
        int (*test_fn)(void) = (int (*)(void))(uintptr_t)sym.val;
        printf("Result: %d\n", test_fn());
    }

    /* "Unload" just means we can reuse the RAM later */
    udynlink_unload_module(&mod);
    pool_offset = 0;   /* reset pool for next boot cycle */

    return 0;
}
```

### With streaming sources

The same pattern works for streaming loads:

```c
uint32_t ram_needed = udynlink_get_ram_requirements_stream(&io, UDYNLINK_LOAD_MODE_COPY_ALL);
if (ram_needed == 0) {
    /* I/O error or invalid signature */
}

void *load_addr = pool_malloc(ram_needed);
udynlink_error_t err = udynlink_load_module_stream(
    &mod, &io, load_addr, ram_needed,
    UDYNLINK_LOAD_MODE_COPY_ALL, work_buf, work_size);
```

### Benefits of pre-computation

- **Deterministic:** Know exactly how much RAM each module consumes before committing.
- **No fragmentation:** Fixed pools eliminate heap fragmentation concerns.
- **Fail-fast:** Detect out-of-memory conditions before starting the load process.

See also [How It Works](how-it-works.md) for a description of what contributes to RAM usage (LOT, `.data`, `.bss`, code, header).
